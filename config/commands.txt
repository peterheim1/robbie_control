# Robbie Voice Commands
# Edit this file to change what Robbie responds to. Restart to reload.
#
# [command name]
# phrases: phrase one, phrase two, ...    plain English, comma-separated
# say:     spoken response
# flags:   cancel_tasks                   optional — cancels any running task
# ros:     type:topic  key=val ...        one or more publish actions
#
# ros types:
#   none
#   empty:/topic                              std_msgs/Empty
#   twist:/topic                              geometry_msgs/Twist  (all zeros)
#   joint_traj:/topic  joint1=rad joint2=rad  trajectory_msgs/JointTrajectory
#   float64_array:/topic  values=v1,v2,...    std_msgs/Float64MultiArray

# ---------------------------------------------------------------------------
# Stop
# ---------------------------------------------------------------------------
# Name must stay "stop_all": the command name is published as the intent
# name on /voice/intent, and bt_executor's intent_cb only handles "stop_all"
# (imperative haltTree + Nav2 goal cancel, 2026-07-13). The ros: zeros below
# are a one-shot reflex only — Nav2 keeps publishing cmd_vel during a
# mission, so the BT cancel is what actually stops the robot.
[stop_all]
phrases: stop, halt, freeze, stop everything, emergency stop
say: stopping
flags: cancel_tasks
ros: empty:/voice/stop
ros: twist:/cmd_vel
ros: float64_array:/drive  values=0,0,0,0

# ---------------------------------------------------------------------------
# Head / look directions — routed through Argus (bt_executor's intent_cb),
# not a direct ros: dispatch. This makes "look" a real current_cmd like
# goto/patrol/dock: it participates in busy-gating (suspends idle fidgets
# while it runs) instead of bypassing the tree entirely.
# ---------------------------------------------------------------------------
[look_right]
phrases: look right, look to the right, turn your head right
say: looking right

[look_left]
phrases: look left, look to the left, turn your head left
say: looking left

[look_up]
phrases: look up, tilt up, tilt your head up
say: looking up

[look_down]
phrases: look down, tilt down, tilt your head down
say: looking down

[look_forward]
phrases: look forward, look front, look straight, look ahead, look straight ahead
say: looking forward

# ---------------------------------------------------------------------------
# Docking
# ---------------------------------------------------------------------------
[dock]
phrases: dock, go dock, start docking, go charge
say: docking
ros: srv_empty:/start_docking

[undock]
phrases: undock, leave the dock, undock yourself
say: undocking
ros: srv_empty:/undock
